fn recombine(token: &[u8], selected_primals: &[&str], nonce: &[u8]) -> VdjKey {
    let mut V = 0;
    for &prim in selected_primals {
        V |= 1 << selected_primals.iter().position(|&p| p == prim).unwrap();
    }

    let mut hasher = Hasher::new();
    hasher.update(token);
    hasher.update(selected_primals.iter().map(|&p| p.as_bytes()).flatten().collect::<Vec<_>>());
    hasher.update(nonce);
    let D = hasher.finalize();

    let mut hasher = Hasher::new();
    hasher.update(token);
    hasher.update(selected_primals.iter().map(|&p| p.as_bytes()).collect::<Vec<_>>().sort_unstable());
    hasher.update(nonce);
    let J = hasher.finalize();

    VdjKey { V, D, J }
}

fn verify(key: &VdjKey, genome_hash: &Hash) -> bool {
    let mut hasher = Hasher::new();
    hasher.update(&key.J);
    hasher.update(genome_hash);
    hasher.finalize() == key.D
}
